PART 5 — Form Validation

Question 13: Validate marks between 0 and 100

PYTHON
class StudentForm(forms.Form):

    name = forms.CharField(
        max_length=100
    )

    marks = forms.IntegerField()


    def clean_marks(self):

        marks = self.cleaned_data[
            "marks"
        ]

        if marks < 0 or marks > 100:

            raise forms.ValidationError(
                "Marks must be between 0 and 100."
            )

        return marks

Question 14: Validate that age is at least 18

PYTHON
def clean_age(self):

    age = self.cleaned_data["age"]

    if age < 18:

        raise forms.ValidationError(
            "Age must be at least 18."
        )

    return age

Question 15: Validate two fields together

Suppose:

TEXT
password
confirm_password

must match.

PYTHON
def clean(self):

    cleaned_data = super().clean()

    password = cleaned_data.get(
        "password"
    )

    confirm = cleaned_data.get(
        "confirm_password"
    )

    if password != confirm:

        raise forms.ValidationError(
            "Passwords do not match."
        )

    return cleaned_data

PYTHON
def student_update(
    request,
    id
):

    student = get_object_or_404(
        Student,
        id=id
    )

    if request.method == "POST":

62. Protect Delete With Permission

This is an excellent interview-level answer:

PYTHON
from django.contrib.auth.decorators import (
    login_required,
    permission_required
)


@login_required
@permission_required(
    "students.delete_student",
    raise_exception=True
)
def student_delete(
    request,
    id
):

    student = get_object_or_404(
        Student,
        id=id
    )

    if request.method == "POST":

        student.delete()

        return redirect(
            "student_list"
        )

    return render(
        request,
        "students/confirm_delete.html",
        {
            "student": student
        }
    )

Now:

TEXT

      ↓

      ↓
Authorization
      ↓
Does user have delete permission?
      ↓
POST?
      ↓
Delete

That's a very strong Django security design.
